The Modulon Programming Language

A language built around modules — small, self-contained pieces of code that snap together into larger programs.

// a module that exposes one function
module Greeter {
    export fn hello(name: Str) -> Str {
        return "hey, ${name}"
    }
}

use Greeter

print(Greeter.hello("world"))
